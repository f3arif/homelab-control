def fetch(url,timeout=2.5):
    try:
        req=urllib.request.Request(url,headers={'User-Agent':'AFZ-Pi-Recovery/1.0'})
        with urllib.request.urlopen(req,timeout=timeout) as r: return True,r.status,r.read(65536).decode('utf-8','replace')
    except Exception as e: return False,0,str(e)
def tcp(host,port,timeout=.8):
    try:
        with socket.create_connection((host,port),timeout): return True
    except Exception: return False
def hp_online():
    ok,code,_=fetch(HP_DIRECT+'/api/hp-health',1.8); return bool(ok and code==200) or tcp(HP_LAN,22)
def wake_hp():
    if hp_online(): return {'ok':True,'state':'already_online'}
    mac=bytes.fromhex(HP_MAC.replace(':','')); pkt=b'\xff'*6+mac*16
    s=socket.socket(socket.AF_INET,socket.SOCK_DGRAM); s.setsockopt(socket.SOL_SOCKET,socket.SO_BROADCAST,1)
    try:
        for _ in range(5): s.sendto(pkt,(BROADCAST,9)); time.sleep(.15)
    finally: s.close()
    return {'ok':True,'state':'wake_sent'}

    def sendj(self,o,c=200):

    def do_GET(self):
        if self.path=='/health': return self.sendj({'ok':True,'service':'afz-ops-pi-recovery','listen':f'{HOST}:{PORT}'})
        if self.path=='/api/status': return self.sendj(status())
        if self.path in ('/','/index.html'):
